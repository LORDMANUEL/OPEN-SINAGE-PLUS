version: '3.8'

services:
  # Backend API
  api:
    build:
      context: ./packages/backend
      dockerfile: ../../docker/Dockerfile.api
    container_name: signage-api
    restart: unless-stopped
    ports:
      - "3000:3000"
    environment:
      NODE_ENV: production
      DATABASE_URL: postgresql://postgres:password@postgres:5432/digital_signage
      REDIS_URL: redis://redis:6379
      JWT_SECRET: ${JWT_SECRET}
      MINIO_ENDPOINT: minio
      MINIO_PORT: 9000
      MINIO_ACCESS_KEY: ${MINIO_ACCESS_KEY}
      MINIO_SECRET_KEY: ${MINIO_SECRET_KEY}
    depends_on:
      - postgres
      - redis
      - minio
    networks:
      - signage-network
    volumes:
      - ./logs/api:/app/logs

  # Admin Panel (Frontend)
  admin:
    build:
      context: ./packages/admin
      dockerfile: ../../docker/Dockerfile.admin
    container_name: signage-admin
    restart: unless-stopped
    ports:
      - "80:80"
      - "443:443"
    environment:
      VITE_API_URL: https://api.tusitio.com
    depends_on:
      - api
    networks:
      - signage-network
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./nginx/ssl:/etc/nginx/ssl:ro

  # Base de Datos PostgreSQL
  postgres:
    image: postgres:15-alpine
    container_name: signage-postgres
    restart: unless-stopped
    ports:
      - "5432:5432"
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: ${DB_PASSWORD}
      POSTGRES_DB: digital_signage
    volumes:
      - postgres-data:/var/lib/postgresql/data
      - ./backups/postgres:/backups
    networks:
      - signage-network
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres"]
      interval: 10s
      timeout: 5s
      retries: 5

  # Redis (Cache y Queue)
  redis:
    image: redis:7-alpine
    container_name: signage-redis
    restart: unless-stopped
    ports:
      - "6379:6379"
    command: redis-server --appendonly yes --requirepass ${REDIS_PASSWORD}
    volumes:
      - redis-data:/data
    networks:
      - signage-network
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5

  # MinIO (Storage S3-compatible)
  minio:
    image: minio/minio:latest
    container_name: signage-minio
    restart: unless-stopped
    ports:
      - "9000:9000"
      - "9001:9001"
    environment:
      MINIO_ROOT_USER: ${MINIO_ACCESS_KEY}
      MINIO_ROOT_PASSWORD: ${MINIO_SECRET_KEY}
    command: server /data --console-address ":9001"
    volumes:
      - minio-data:/data
    networks:
      - signage-network
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:9000/minio/health/live"]
      interval: 30s
      timeout: 20s
      retries: 3

  # Prometheus (Monitoring)
  prometheus:
    image: prom/prometheus:latest
    container_name: signage-prometheus
    restart: unless-stopped
    ports:
      - "9090:9090"
    volumes:
      - ./monitoring/prometheus.yml:/etc/prometheus/prometheus.yml:ro
      - prometheus-data:/prometheus
    command:
      - '--config.file=/etc/prometheus/prometheus.yml'
      - '--storage.tsdb.path=/prometheus'
    networks:
      - signage-network

  # Grafana (Dashboard)
  grafana:
    image: grafana/grafana:latest
    container_name: signage-grafana
    restart: unless-stopped
    ports:

      GF_SECURITY_ADMIN_PASSWORD: ${GRAFANA_PASSWORD}
      GF_INSTALL_PLUGINS: grafana-clock-panel

      - ./monitoring/grafana/datasources:/etc/grafana/provisioning/datasources:ro

      - signage-network

      - loki-data:/loki
    networks:
      - signage-network

FROM node:20-alpine AS builder

WORKDIR /app

# Copiar archivos de dependencias

RUN npm ci --only=production

# Copiar código fuente
COPY src ./src
COPY prisma ./prisma

# Generar Prisma Client
RUN npx prisma generate

# Build TypeScript
RUN npm run build

# Stage de producción
FROM node:20-alpine

WORKDIR /app

# Copiar dependencias y build
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/dist ./dist
COPY --from=builder /app/prisma ./prisma
COPY package*.json ./

# Crear usuario no-root
RUN addgroup -g 1001 -S nodejs && \
    adduser -S nodejs -u 1001

USER nodejs

EXPOSE 3000

# Health check
HEALTHCHECK --interval=30s --timeout=3s --start-period=40s \
  CMD node -e "require('http').get('http://localhost:3000/health', (r) => {process.exit(r.statusCode === 200 ? 0 : 1)})"

CMD ["node", "dist/app.js"]

# ==========================================
# NGINX CONFIGURATION
# ==========================================
# nginx/nginx.conf

user nginx;
worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for"';

    access_log /var/log/nginx/access.log main;

    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;

    # Gzip compression
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types text/plain text/css text/xml text/javascript 
               application/json application/javascript application/xml+rss 
               application/rss+xml font/truetype font/opentype 
               application/vnd.ms-fontobject image/svg+xml;

    # Servidor Admin Panel
    server {
        listen 80;
        listen [::]:80;
        server_name tusitio.com www.tusitio.com;

        # Redirigir a HTTPS
        return 301 https://$server_name$request_uri;
    }

    server {
        listen 443 ssl http2;
        listen [::]:443 ssl http2;
        server_name tusitio.com www.tusitio.com;

        # SSL Configuration
        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!MD5;
        ssl_prefer_server_ciphers on;

        # Security headers
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-XSS-Protection "1; mode=block" always;
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

        root /usr/share/nginx/html;
        index index.html;

        # Admin Panel
        location / {
            try_files $uri $uri/ /index.html;
        }

        # API Proxy
        location /api {
            proxy_pass http://api:3000;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection 'upgrade';
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_cache_bypass $http_upgrade;
            proxy_read_timeout 300s;
            proxy_connect_timeout 75s;
        }

        # WebSocket
        location /socket.io {
            proxy_pass http://api:3000;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        # Cache static assets
        location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
            expires 1y;
            add_header Cache-Control "public, immutable";
        }
    }
}

# GITHUB ACTIONS - CI/CD

# .github/workflows/deploy.yml

name: Deploy Digital Signage

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}

jobs:
  # Test Backend
  test-backend:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      
      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '20'
          cache: 'npm'
      
      - name: Install dependencies
        run: |
          cd packages/backend
          npm ci
      
      - name: Run linter
        run: |
          cd packages/backend
          npm run lint
      
      - name: Run tests
        run: |
          cd packages/backend
          npm run test
      
      - name: Generate Prisma Client
        run: |
          cd packages/backend
          npx prisma generate

  # Build and Push Docker Images
  build-and-push:
    needs: test-backend
    runs-on: ubuntu-latest
    if: github.event_name == 'push'
    steps:
      - uses: actions/checkout@v3
      
      - name: Log in to Container Registry
        uses: docker/login-action@v2
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}
      
      - name: Build and push API image
        uses: docker/build-push-action@v4
        with:
          context: ./packages/backend
          file: ./docker/Dockerfile.api
          push: true
          tags: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}/api:latest
      
      - name: Build and push Admin image
        uses: docker/build-push-action@v4
        with:
          context: ./packages/admin
          file: ./docker/Dockerfile.admin
          push: true
          tags: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}/admin:latest

  # Deploy to Production
  deploy:
    needs: build-and-push
    runs-on: ubuntu-latest
    if: github.event_name == 'push' && github.ref == 'refs/heads/main'
    steps:
      - name: Deploy to server
        uses: appleboy/ssh-action@master
        with:
          host: ${{ secrets.SERVER_HOST }}
          username: ${{ secrets.SERVER_USER }}
          key: ${{ secrets.SSH_PRIVATE_KEY }}
          script: |
            cd /opt/digital-signage
            docker-compose pull
            docker-compose up -d
            docker system prune -af

  # Build Electron Player
  build-electron:
    runs-on: ${{ matrix.os }}
    strategy:
      matrix:
        os: [macos-latest, ubuntu-latest, windows-latest]
    steps:
      - uses: actions/checkout@v3
      
      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '20'
      
      - name: Install dependencies
        run: |
          cd packages/player
          npm ci
      
      - name: Build Electron app
        run: |
          cd packages/player
          npm run build
          npm run package
      
      - name: Upload artifacts
        uses: actions/upload-artifact@v3
        with:
          name: electron-${{ matrix.os }}
          path: packages/player/release/*

# install.sh

#!/bin/bash
set -e

echo "🚀 Instalando Digital Signage Enterprise..."

# Verificar requisitos
command -v docker >/dev/null 2>&1 || { echo "Docker no está instalado. Abortando." >&2; exit 1; }
command -v docker-compose >/dev/null 2>&1 || { echo "Docker Compose no está instalado. Abortando." >&2; exit 1; }

# Crear directorios
mkdir -p logs/{api,nginx}
mkdir -p backups/{postgres,app}
mkdir -p nginx/ssl
mkdir -p monitoring/{prometheus,grafana/{dashboards,datasources},loki}

# Copiar archivos de configuración
if [ ! -f .env ]; then
    cp .env.example .env
    echo "⚠️  Archivo .env creado. Por favor, configúralo antes de continuar."
    exit 1
fi

# Generar SSL self-signed (para desarrollo)
if [ ! -f nginx/ssl/cert.pem ]; then
    echo "🔐 Generando certificados SSL..."
    openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
        -keyout nginx/ssl/key.pem \
        -out nginx/ssl/cert.pem \
        -subj "/C=US/ST=State/L=City/O=Organization/CN=localhost"
fi

# Iniciar servicios
echo "📦 Iniciando contenedores..."
docker-compose -f docker/docker-compose.prod.yml up -d

# Esperar a que Postgres esté listo
echo "⏳ Esperando a PostgreSQL..."
sleep 10

# Ejecutar migraciones
echo "🔄 Ejecutando migraciones..."
docker-compose -f docker/docker-compose.prod.yml exec api npx prisma migrate deploy

# Crear usuario admin inicial
echo "👤 Creando usuario administrador..."
docker-compose -f docker/docker-compose.prod.yml exec api node scripts/create-admin.js

echo ""
echo "✅ Instalación completada!"
echo ""
echo "🌐 URLs de acceso:"
echo "   Admin Panel: https://localhost"
echo "   API: http://localhost:3000"
echo "   Grafana: http://localhost:3001"
echo "   MinIO Console: http://localhost:9001"
echo ""
echo "📖 Documentación: https://docs.tusitio.com"
echo ""

# ==========================================
# FIN DEL ARCHIVO
